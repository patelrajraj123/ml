import numpy as np 
import matplotlib.pyplot as plt 
 
def unitstep(v): 
    return 1 if v >= 0 else 0 
 
def perceptronModel(x, w, b): 
    v = np.dot(w, x) + b 
    y = unitstep(v) 
    return y 
 
def NOT_LogicFunction(x): 
    w = np.array([-1]) 
    b = 0.5 
    return perceptronModel(x, w, b) 
 
inputs = np.array([[0], [1]]) 
 
outputs = [NOT_LogicFunction(x) for x in inputs] 
 
x = inputs[:, 0] 
y = outputs 
 
plt.figure(figsize=(6, 4)) 
plt.scatter(x, y, c=outputs, cmap="viridis", s=100, edgecolors='k') 
 
plt.xlabel('Input') 
plt.ylabel('Output') 
plt.title('NOT logic function with w=[-1] and b=0.5') 
plt.xticks([0, 1]) 
plt.yticks([0, 1]) 
 
plt.plot([0.5, 0.5], [-0.5, 1.5], 'r--', label='Decision Boundary (x=0.5)') 
 
plt.xlim(-0.5, 1.5) 
plt.ylim(-0.5, 1.5) 
plt.grid(True) 
plt.legend() 
plt.show()
