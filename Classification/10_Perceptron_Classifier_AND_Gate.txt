import numpy as np 
import matplotlib.pyplot as plt 
 
def unitstep(v): 
    return 1 if v >= 0 else 0 
 
def perceptronModel(x, w, b): 
    v = np.dot(w, x) + b 
    y = unitstep(v) 
    return y 
 
def AND_LogicFunction(x): 
    w = np.array([1.5, 1.5]) 
    b = -2.25 
    return perceptronModel(x, w, b) 
 
inputs = np.array([[0, 0], [0, 1], [1, 0], [1, 1]]) 
 
outputs = [AND_LogicFunction(x) for x in inputs] 
 
x = inputs[:, 0] 
y = inputs[:, 1] 
 
plt.figure(figsize=(6, 6)) 
plt.scatter(x, y, c=outputs, cmap="viridis", s=100, edgecolors='k') 
 
plt.xlabel('input 1') 
plt.ylabel('input 2') 
plt.title('AND logic function with w=(1.5, 1.5) and b=-2.25') 
plt.xticks([0, 1]) 
plt.yticks([0, 1]) 
 
plt.plot([0.75, 0.75], [-0.5, 1.5], 'r--') 
plt.plot([-0.5, 1.5], [0.75, 0.75], 'b--') 
 
plt.xlim(-0.5, 1.5) 
plt.ylim(-0.5, 1.5) 
plt.grid(True) 
plt.show()
