import pandas as pd 
from sklearn.model_selection import train_test_split 
from sklearn.metrics import accuracy_score 
import matplotlib.pyplot as plt 
from sklearn.tree import plot_tree 
from sklearn.naive_bayes import GaussianNB 
from sklearn.metrics import mean_squared_error 
import numpy as np 
 
dataset_fruit = { 
    'Colour': ['Red', 'Orange', 'Red', 'Green', 'Orange', 'Green', 'Red', 'Orange', 'Red', 'Green'], 
    'Shape': ['Round', 'Round', 'Oval', 'Round', 'Oval', 'Round', 'Oval', 'Round', 'Round', 'Oval'], 
    'Taste': ['Sweet', 'Sweet', 'Sweet', 'Sour', 'Sweet', 'Sweet', 'Sour', 'Sour', 'Sweet', 'Sour'], 
    'Label': ['Apple', 'Orange', 'Apple', 'Apple', 'Orange', 'Apple', 'Apple', 'Orange', 'Apple', 'Apple'] 
} 
 
df = pd.DataFrame(dataset_fruit) 
print(df) 
 
col_map={'Red': 0, 'Orange': 1, 'Green': 2} 
df['Colour'] = df['Colour'].map(col_map) 
 
sh_map={'Round': 0, 'Oval': 1} 
df['Shape'] = df['Shape'].map(sh_map) 
 
tas_map={'Sweet': 0, 'Sour': 1} 
df['Taste'] = df['Taste'].map(tas_map) 
 
lab_map={'Apple': 0, 'Orange': 1} 
df['Label'] = df['Label'].map(lab_map) 
 
x = df.drop('Label', axis=1) 
y = df['Label'] 
 
print(x) 
print(y) 
print(list(x.columns)) 
 
x_train, x_test, y_train, y_test = train_test_split(x, y, test_size=0.2, random_state=42) 
 
model = GaussianNB() 
model.fit(x_train, y_train) 
 
y_pred = model.predict(x_test) 
print( y_pred) 
accuracy = accuracy_score(y_test, y_pred)
